# dconf keys Night City writes: "<key> <GVariant value>", one per line.
# backup.sh records the current value of each key (or that it is unset), install.sh writes
# these values, and uninstall.sh puts back exactly what was there (or resets the key).
# Extension keys can be written before the extension is installed; dconf doesn't need the schema.

# ---- Appearance ----
/org/gnome/desktop/interface/color-scheme 'prefer-dark'
/org/gnome/desktop/interface/accent-color 'yellow'
/org/gnome/desktop/interface/gtk-theme 'adw-gtk3-dark'
/org/gnome/desktop/interface/icon-theme 'Papirus-Dark'
/org/gnome/desktop/interface/cursor-theme 'Bibata-Modern-Amber'
